import { inject } from '@angular/core';
import { CanActivateFn, Router } from '@angular/router';
import { RolUsuario } from '../../models/consultorio.model';
import { AuthService } from '../services/auth.service';

export const authGuard: CanActivateFn = () => {
  const auth = inject(AuthService);
  const router = inject(Router);

  if (auth.estaAutenticado()) {
    return true;
  }

  return router.createUrlTree(['/login']);
};

export function roleGuard(rolesPermitidos: RolUsuario[]): CanActivateFn {
  return () => {
    const auth = inject(AuthService);
    const router = inject(Router);

    if (!auth.estaAutenticado()) {
      return router.createUrlTree(['/login']);
    }

    if (auth.tieneRol(...rolesPermitidos)) {
      return true;
    }

    return router.createUrlTree(['/tabs/sala']);
  };
}

  readonly usuarioActual = signal<UsuarioSesion | null>(this.cargarSesionGuardada());
  readonly esSuperAdmin = computed(() => this.usuarioActual()?.rol === 'super_admin');
  readonly esDoctor = computed(() => this.usuarioActual()?.rol === 'doctor');
  readonly esAsistente = computed(() => this.usuarioActual()?.rol === 'asistente');

  private normalizarRol(raw: unknown): RolUsuario | null {
    const val = String(raw ?? '').trim().toLowerCase();
    if (val === 'super_admin' || val === 'admin') return 'super_admin';
    if (val === 'doctor') return 'doctor';
    if (val === 'asistente') return 'asistente';
    return null;
  }

  private cargarSesionGuardada(): UsuarioSesion | null {
    try {
      const raw = localStorage.getItem(this.STORAGE_KEY);
      if (!raw) return null;
      const parsed = JSON.parse(raw) as UsuarioSesion;
      const rolNormalizado = this.normalizarRol(parsed.rol);
      if (!rolNormalizado) return null;
      return {
        ...parsed,
        rol: rolNormalizado,
        rol_etiqueta: parsed.rol_etiqueta || ETIQUETAS_ROL[rolNormalizado],
      };
    } catch {
      return null;
    }
  }

  estaAutenticado(): boolean {
    return this.usuarioActual() !== null;
  }

  tieneRol(...roles: RolUsuario[]): boolean {
    const actual = this.usuarioActual()?.rol;
    return !!actual && roles.includes(actual);
  }

  async iniciarSesion(email: string, password: string): Promise<RespuestaLogin> {
    const correoLimpio = email.trim().toLowerCase();
    if (!correoLimpio || !password) {
      return { ok: false, mensaje: 'Ingresa tu correo electrÃ³nico y contraseÃ±a.' };
    }

    try {
      const resp = await fetch(this.API_LOGIN_URL, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Accept: 'application/json',
        },
        body: JSON.stringify({
          email: correoLimpio,
          password,
        }),
      });

      const data = await resp.json().catch(() => ({}));

      if (!resp.ok || !data?.ok) {
        return {
          ok: false,
          mensaje: data?.mensaje || data?.message || 'Correo o contraseÃ±a incorrectos.',
        };
      }

      const u = data.user;
      if (Number(u.activo) === 0) {
        return {
          ok: false,
          mensaje: 'Tu cuenta estÃ¡ desactivada. Contacta a la administradora.',
        };
      }

      const rolNormalizado = this.normalizarRol(u.rol);
      if (!rolNormalizado) {
        return {
          ok: false,
          mensaje: 'Acceso exclusivo para personal autorizado de la clÃ­nica.',
        };
      }

      const sesion: UsuarioSesion = {
        id: u.id,
        nombre: `${u.nombre} ${u.apellido || ''}`.trim(),
        apellido: u.apellido || '',
        email: u.email,
        rol: rolNormalizado,
        rol_etiqueta: u.rol_etiqueta || ETIQUETAS_ROL[rolNormalizado],
        foto_perfil: u.foto_perfil ?? null,
      };

      this.usuarioActual.set(sesion);
      localStorage.setItem(this.STORAGE_KEY, JSON.stringify(sesion));
      return { ok: true };
    } catch {
      return {
        ok: false,
        mensaje: 'No se pudo conectar con el servidor de Dentissa. Verifica tu conexiÃ³n a internet.',
      };
    }
  }

  cerrarSesion(): void {
    this.usuarioActual.set(null);
    localStorage.removeItem(this.STORAGE_KEY);
  }
}

  sincronizarDesdeBackend(): void {
    if (!this.historialDisponible()) {

    this.http.get<any>(`${this.apiIotUrl}/estado`).subscribe({
      next: (res) => {

        this.esp32EnLinea.set(online);

        if (res?.entradas_qr_hoy !== undefined) {
          this.entradasQrHoyBackend.set(Number(res.entradas_qr_hoy));
        }

        if (Array.isArray(res.asientos)) {
          const ahora = Date.now();
          this.asientos.update((actuales) =>
            actuales.map((a) => {
              const remoto = res.asientos.find((r: any) => Number(r.numero) === a.numero);
              if (!remoto) return a;
              const ocupadoAhora = Boolean(remoto.ocupado);
              const incremento = !a.ocupado && ocupadoAhora ? 1 : 0;

              if (!a.ocupado && ocupadoAhora) {
                this.inicioOcupacionMs.set(a.numero, ahora);
              } else if (a.ocupado && !ocupadoAhora) {
                const inicio = this.inicioOcupacionMs.get(a.numero);
                if (inicio) {
                  const duracion = Math.max(1, Math.floor((ahora - inicio) / 1000));
                  this.segundosAcumulados.update((prev) => ({
                    ...prev,
                    [a.numero]: (prev[a.numero] || 0) + duracion,
                  }));
                  this.inicioOcupacionMs.delete(a.numero);
                }
              }

              if (a.ocupado !== ocupadoAhora) {
                if (!this.luces().encendido) {
                  this.cambiarModoActuador('LUCES', 'MANUAL_ON', false);
                }
                a.zonasVentilador.forEach((z) => {
                  const ventId = (`VENTILADOR_${z}`) as DispositivoId;
                  const vent = this.ventiladores().find((v) => v.id === ventId);
                  if (vent && vent.modo !== 'AUTO') {
                    this.cambiarModoActuador(ventId, 'AUTO', false);
                  }
                });
              }

              return {
                ...a,
                ocupado: ocupadoAhora,
                vecesUsado: remoto.vecesUsado ?? a.vecesUsado + incremento,
              };
            })
          );
          this.aforoActual.set(this.asientosOcupados());
        }

        if (Array.isArray(res.ventiladores)) {
          this.ventiladores.update((actuales) =>
            actuales.map((v, idx) => {
              const remoto = res.ventiladores.find((r: any) => Number(r.numero) === idx + 1);
              if (!remoto) return v;
              return {
                ...v,
                modo: (remoto.modo as ModoControl) || v.modo,
                encendido: Boolean(remoto.encendido),
              };
            })
          );
        }

        if (res.luces) {
          this.luces.update((l) => ({
            ...l,
            modo: (res.luces.modo as ModoControl) || l.modo,
            encendido: Boolean(res.luces.encendido),
          }));
        }

        this.ultimaActualizacion.set(new Date());
      },
      error: () => {
        this.esp32EnLinea.set(false);
      },
    });
  }

  alternarActuadorConRetornoAuto(id: DispositivoId): void {

  cambiarModoActuador(id: DispositivoId, nuevoModo: ModoControl, programarRetornoAuto = true): void {
    this.bloqueandoSync = true;
    this.limpiarTemporizadorAuto(id);

    if (id === 'LUCES') {
      const encendido =
        nuevoModo === 'MANUAL_ON'
          ? true
          : nuevoModo === 'MANUAL_OFF'
            ? false
            : this.aforoActual() > 0 || this.puertaAbierta();

      this.luces.update((l) => ({ ...l, modo: nuevoModo, encendido }));
      this.registrarAuditoria('Tira LED Sala', `Modo ${nuevoModo}`, 'MANUAL');
      this.enviarComandoActuador('LUCES', nuevoModo, encendido);
    } else {
      let estadoEncendido = false;
      this.ventiladores.update((lista) =>
        lista.map((v) => {
          if (v.id !== id) return v;
          const zonaNum = Number(id.split('_')[1]);
          const hayPresencia = this.asientos().some(
            (a) => a.ocupado && a.zonasVentilador.includes(zonaNum)
          );
          estadoEncendido =
            nuevoModo === 'MANUAL_ON'
              ? true
              : nuevoModo === 'MANUAL_OFF'
                ? false
                : hayPresencia;

          return { ...v, modo: nuevoModo, encendido: estadoEncendido };
        })
      );

      const vent = this.ventiladores().find((v) => v.id === id);
      this.registrarAuditoria(vent?.nombre || id, `Modo ${nuevoModo}`, 'MANUAL');
      this.enviarComandoActuador(id, nuevoModo, estadoEncendido);
    }

    if (programarRetornoAuto && nuevoModo !== 'AUTO') {
      this.programarRegresoAuto(id);
    }
  }

  private programarRegresoAuto(id: DispositivoId): void {
    let restantes = this.tiempoRetornoAutoSeg;
    this.segundosRestantesAuto.update((map) => ({ ...map, [id]: restantes }));

    const intervalo = setInterval(() => {
      restantes -= 1;
      if (restantes > 0) {
        this.segundosRestantesAuto.update((map) => ({ ...map, [id]: restantes }));
      } else {
        this.limpiarTemporizadorAuto(id);
      }
    }, 1000);

    const timeout = setTimeout(() => {
      this.limpiarTemporizadorAuto(id);
      this.cambiarModoActuador(id, 'AUTO', false);
    }, this.tiempoRetornoAutoSeg * 1000);

    this.intervalosCuenta.set(id, intervalo);
    this.timersAuto.set(id, timeout);
  }

  private limpiarTemporizadorAuto(id: DispositivoId): void {
    const t = this.timersAuto.get(id);
    if (t) clearTimeout(t);
    this.timersAuto.delete(id);

    const i = this.intervalosCuenta.get(id);
    if (i) clearInterval(i);
    this.intervalosCuenta.delete(id);

    this.segundosRestantesAuto.update((map) => {
      const copia = { ...map };
      delete copia[id];
      return copia;
    });
  }

  restablecerTodoAuto(): void {
    this.cambiarModoActuador('LUCES', 'AUTO', false);
    this.ventiladores().forEach((v) => this.cambiarModoActuador(v.id, 'AUTO', false));
  }

  apagarTodo(): void {
    this.cambiarModoActuador('LUCES', 'MANUAL_OFF', false);
    this.ventiladores().forEach((v) => this.cambiarModoActuador(v.id, 'MANUAL_OFF', false));
    this.registrarAuditoria('Sala Completa', 'Apagado general (Leds y ventiladores)', 'MANUAL');
  }

  private enviarComandoActuador(dispositivo: DispositivoId, modo: ModoControl, encendido: boolean): void {
    this.http
      .post(`${this.apiIotUrl}/actuadores`, {
        dispositivo,
        modo,
        encendido,
        ventiladores: this.ventiladores().map((v, i) => ({
          numero: i + 1,
          modo: v.modo,
          encendido: v.encendido,
        })),
      })
      .subscribe({
        next: () => (this.bloqueandoSync = false),
        error: () => (this.bloqueandoSync = false),
      });
  }

  alternarPuertaManual(): void {
    const abrir = !this.puertaAbierta();
    this.puertaAbierta.set(abrir);
    if (this.puertaTimer) clearTimeout(this.puertaTimer);

    if (abrir) {
      this.http.post(`${this.apiIotUrl}/puerta/abrir`, { codigo_qr: 'MANUAL_APP' }).subscribe({
        error: () => {},
      });
      if (!this.luces().encendido) {
        this.cambiarModoActuador('LUCES', 'MANUAL_ON', false);
      }
      this.recalcularAutomatizacion();
      this.registrarAuditoria('Puerta Servomotor', 'Apertura manual desde app', 'MANUAL');

      this.puertaTimer = setTimeout(() => {
        this.puertaAbierta.set(false);
        this.recalcularAutomatizacion();
      }, 3000);
    } else {
      this.recalcularAutomatizacion();
      this.registrarAuditoria('Puerta Servomotor', 'Cierre manual desde app', 'MANUAL');
    }
  }
  private recalcularAutomatizacion(): void {
    const hayPersonas = this.aforoActual() > 0 || this.asientosOcupados() > 0;
    const puerta = this.puertaAbierta();

    if (hayPersonas && !this.luces().encendido && this.luces().modo === 'AUTO') {
      this.cambiarModoActuador('LUCES', 'MANUAL_ON', false);
    }

    const asientosActuales = this.asientos();
    this.ventiladores.update((lista) =>
      lista.map((v) => {
        if (v.modo !== 'AUTO') return v;
        const zonaNum = Number(v.id.split('_')[1]);
        const ocupadoEnZona = asientosActuales.some(
          (a) => a.ocupado && a.zonasVentilador.includes(zonaNum)
        );
        return { ...v, encendido: ocupadoEnZona };
      })
    );

    this.ultimaActualizacion.set(new Date());
  }

  private registrarAuditoria(dispositivo: string, accion: string, origen: OrigenMovimiento): void {
    const user = this.auth.usuarioActual();
    this.auditoria.update((lista) => [
      {
        id: this.seqId++,
        usuarioNombre: origen === 'AUTOMATICO' ? 'Sistema IoT' : user?.nombre || 'Personal',
        rol: user?.rol || 'super_admin',
        dispositivo,
        accion,
        origen,
        fechaHora: new Date(),
      },
      ...lista.slice(0, 29),
    ]);
  }
}

import { Injectable } from '@angular/core';

@Injectable({
  providedIn: 'root',
})
export class TotpService {
  readonly PERIODO_SEGUNDOS = 30;

  obtenerSegundosRestantes(): number {
    const epochSegundos = Math.floor(Date.now() / 1000);
    return this.PERIODO_SEGUNDOS - (epochSegundos % this.PERIODO_SEGUNDOS);
  }

  async generarCodigo(secretBase32: string, offsetVentana = 0): Promise<string> {
    const keyBuffer = this.base32ToArrayBuffer(secretBase32);
    const counter =
      Math.floor(Date.now() / 1000 / this.PERIODO_SEGUNDOS) + offsetVentana;

    const counterBuffer = new ArrayBuffer(8);
    const view = new DataView(counterBuffer);
    view.setUint32(0, Math.floor(counter / 0x100000000), false);
    view.setUint32(4, counter >>> 0, false);

    const cryptoKey = await crypto.subtle.importKey(
      'raw',
      keyBuffer,
      { name: 'HMAC', hash: 'SHA-1' },
      false,
      ['sign']
    );

    const signature = await crypto.subtle.sign('HMAC', cryptoKey, counterBuffer);
    const hmac = new Uint8Array(signature);

    const offset = hmac[hmac.length - 1] & 0x0f;
    const binary =
      ((hmac[offset] & 0x7f) << 24) |
      ((hmac[offset + 1] & 0xff) << 16) |
      ((hmac[offset + 2] & 0xff) << 8) |
      (hmac[offset + 3] & 0xff);

    return (binary % 1000000).toString().padStart(6, '0');
  }

  async generarPayloadQr(userId: string, secretBase32: string): Promise<{
    payload: string;
    codigoOtp: string;
  }> {
    const codigoOtp = await this.generarCodigo(secretBase32, 0);
    return {
      payload: `DENTISSA:${userId}:${codigoOtp}`,
      codigoOtp,
    };
  }

  async verificarCodigo(secretBase32: string, codigoRecibido: string): Promise<boolean> {
    const limpio = codigoRecibido.trim();
    if (!/^\d{6}$/.test(limpio)) return false;

    for (const ventana of [0, -1, 1]) {
      const esperado = await this.generarCodigo(secretBase32, ventana);
      if (esperado === limpio) {
        return true;
      }
    }
    return false;
  }

  private base32ToArrayBuffer(base32: string): ArrayBuffer {
    const alphabet = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567';
    const clean = base32.toUpperCase().replace(/=+$/, '').replace(/\s+/g, '');
    let bits = 0;
    let value = 0;
    const output: number[] = [];

    for (let i = 0; i < clean.length; i++) {
      const idx = alphabet.indexOf(clean[i]);
      if (idx === -1) continue;
      value = (value << 5) | idx;
      bits += 5;
      if (bits >= 8) {
        output.push((value >>> (bits - 8)) & 255);
        bits -= 8;
      }
    }
    return new Uint8Array(output).buffer;
  }
}

export class AccesoQrPage implements OnInit, OnDestroy {
  qrService = inject(CitasQrService);
  auth = inject(AuthService);
  totp = inject(TotpService);

  modoVista: 'mi-qr' | 'escaner' = 'mi-qr';
  codigoManual = '';

  qrImagenUrl = signal<string>('');
  codigoOtpActual = signal<string>('------');
  payloadQrActual = signal<string>('');
  segundosRestantes = signal<number>(30);
  resultadoValidacion = signal<{ valido: boolean; mensaje: string } | null>(null);

  private intervaloTimer: ReturnType<typeof setInterval> | null = null;
  private ultimaVentanaGenerada = -1;

  constructor() {
    addIcons({
      cameraOutline,
      keyOutline,
      checkmarkCircleOutline,
      closeCircleOutline,
      shieldCheckmarkOutline,
      timeOutline,
    });
  }

  async ngOnInit(): Promise<void> {
    await this.actualizarCicloTotp(true);
    this.intervaloTimer = setInterval(() => {
      void this.actualizarCicloTotp(false);
    }, 1000);
  }

  ngOnDestroy(): void {
    if (this.intervaloTimer) {
      clearInterval(this.intervaloTimer);
    }
  }

  private async actualizarCicloTotp(forzar: boolean): Promise<void> {
    const restantes = this.totp.obtenerSegundosRestantes();
    this.segundosRestantes.set(restantes);

    const ventanaActual = Math.floor(Date.now() / 1000 / this.totp.PERIODO_SEGUNDOS);
    if (!forzar && ventanaActual === this.ultimaVentanaGenerada) {
      return;
    }

    this.ultimaVentanaGenerada = ventanaActual;
    const user = this.auth.usuarioActual();
    if (!user) return;

    const { payload, codigoOtp } = await this.totp.generarPayloadQr(
      user.id,
      user.totpSecret
    );

    this.codigoOtpActual.set(codigoOtp);
    this.payloadQrActual.set(payload);
    this.codigoManual = payload;

    const url = await QRCode.toDataURL(payload, { width: 250, margin: 1 });
    this.qrImagenUrl.set(url);
  }

  async escanearConCamara(): Promise<void> {
    try {
      const { barcodes } = await BarcodeScanner.scan();
      if (barcodes.length > 0 && barcodes[0].rawValue) {
        this.codigoManual = barcodes[0].rawValue;
        await this.activarPuertaConQr(this.codigoManual);
      }
    } catch {
      this.resultadoValidacion.set({
        valido: false,
        mensaje:
          'Cámara nativa no disponible en navegador web. Usa el campo o la lista inferior.',
      });
    }
  }

  async simularEscaneoEmpleado(usuario: UsuarioSesion): Promise<void> {
    const { payload } = await this.totp.generarPayloadQr(
      usuario.id,
      usuario.totpSecret
    );
    this.codigoManual = payload;
    await this.activarPuertaConQr(payload);
  }

  async activarPuertaConQr(codigo: string): Promise<void> {
    if (!codigo.trim()) return;
    const res = await this.qrService.validarQrPuerta(codigo);
    this.resultadoValidacion.set({
      valido: res.valido,
      mensaje: res.mensaje,
    });
  }
}

export class QrAccesoPage implements OnInit, OnDestroy {
  readonly auth = inject(AuthService);
  readonly etiquetaRol = computed(() => {
    const u = this.auth.usuarioActual();
    if (u?.rol_etiqueta) return u.rol_etiqueta;
    if (u?.rol === 'super_admin') return 'Doctora titular';
    if (u?.rol === 'doctor') return 'Doctor';
    return 'Asistente';
  });
  private consultorio = inject(ConsultorioService);
  private html5Qr: Html5Qrcode | null = null;
  private procesando = false;
  private timerInterval: ReturnType<typeof setInterval> | null = null;
  private ultimoMinutoGenerado = -1;

  vistaActiva = signal<'credencial' | 'escaner'>('credencial');
  qrDataUrl = signal<string>('');
  segundosRestantes = signal<number>(60);

  escaneando = signal(false);
  estadoMensaje = signal('');
  estadoTipo = signal<'ok' | 'err' | ''>('');

  constructor() {
    addIcons({
      qrCodeOutline,
      cameraOutline,
      stopCircleOutline,
      checkmarkCircleOutline,
      closeCircleOutline,
      timeOutline,
    });
  }

  ngOnInit(): void {
    this.actualizarCicloQr();
    this.timerInterval = setInterval(() => {
      this.actualizarCicloQr();
    }, 1000);
  }

  private async actualizarCicloQr(): Promise<void> {
    const ahoraSeg = Math.floor(Date.now() / 1000);
    const minutoActual = Math.floor(ahoraSeg / 60);
    const restantes = 60 - (ahoraSeg % 60);
    this.segundosRestantes.set(restantes);

    if (minutoActual !== this.ultimoMinutoGenerado) {
      this.ultimoMinutoGenerado = minutoActual;
      await this.generarQrDinamico(minutoActual);
    }
  }

  private async generarQrDinamico(ventanaMinuto: number): Promise<void> {
    const user = this.auth.usuarioActual();
    const uid = Number(user?.id) || 0;
    const rol = (user?.rol ?? 'staff').toUpperCase();
    const firma = ((uid * 7919 + ventanaMinuto * 104729) >>> 0).toString(16).toUpperCase().padStart(8, '0');
    const payload = `DENTISSA-STAFF:${uid}:${rol}:${ventanaMinuto}:${firma}`;

    try {
      const url = await QRCode.toDataURL(payload, {
        width: 260,
        margin: 1,
        color: {
          dark: '#22181a',
          light: '#ffffff',
        },
      });
      this.qrDataUrl.set(url);
    } catch {}
  }

  async cambiarVista(event: CustomEvent): Promise<void> {
    const nuevaVista = event.detail?.value as 'credencial' | 'escaner';
    if (!nuevaVista || nuevaVista === this.vistaActiva()) return;

    if (nuevaVista === 'credencial' && this.escaneando()) {
      await this.detenerCamara();
    }
    this.vistaActiva.set(nuevaVista);
  }

  async iniciarCamara(): Promise<void> {
    this.estadoMensaje.set('');
    this.estadoTipo.set('');

    try {
      this.html5Qr = new Html5Qrcode('qr-reader');
      this.escaneando.set(true);

      await this.html5Qr.start(
        { facingMode: 'environment' },
        { fps: 10, qrbox: { width: 220, height: 220 } },
        async (textoLeido) => {
          if (this.procesando) return;
          this.procesando = true;

          const resultado = await this.consultorio.validarYAbrirPuertaQr(textoLeido);
          this.estadoTipo.set(resultado.ok ? 'ok' : 'err');
          this.estadoMensaje.set(resultado.mensaje);

          setTimeout(() => {
            this.procesando = false;
          }, 3000);
        },
        () => {}
      );
    } catch {
      this.escaneando.set(false);
      this.estadoTipo.set('err');
      this.estadoMensaje.set('No se pudo acceder a la cÃ¡mara. Verifica los permisos del navegador.');
    }
  }

  async detenerCamara(): Promise<void> {
    if (this.html5Qr && this.escaneando()) {
      try {
        await this.html5Qr.stop();
        this.html5Qr.clear();
      } catch {}
    }
    this.escaneando.set(false);
  }

  ngOnDestroy(): void {
    if (this.timerInterval) {
      clearInterval(this.timerInterval);
      this.timerInterval = null;
    }
    this.detenerCamara();
  }

export class SalaPage implements OnInit {
  auth = inject(AuthService);
  consultorio = inject(ConsultorioService);
  private router = inject(Router);
  private route = inject(ActivatedRoute);

  vistaActiva = signal<'plano' | 'stats' | 'auditoria'>('plano');
  fotoError = signal<boolean>(false);

  readonly mesesLista = [
    { valor: 1, nombre: 'Enero' },
    { valor: 2, nombre: 'Febrero' },
    { valor: 3, nombre: 'Marzo' },
    { valor: 4, nombre: 'Abril' },
    { valor: 5, nombre: 'Mayo' },
    { valor: 6, nombre: 'Junio' },
    { valor: 7, nombre: 'Julio' },
    { valor: 8, nombre: 'Agosto' },
    { valor: 9, nombre: 'Septiembre' },
    { valor: 10, nombre: 'Octubre' },
    { valor: 11, nombre: 'Noviembre' },
    { valor: 12, nombre: 'Diciembre' },
  ];

  readonly aniosLista: number[] = (() => {
    const actual = new Date().getFullYear();
    return [actual, actual - 1, actual - 2, actual - 3];
  })();

  ventiladoresActivos = computed(
    () => this.consultorio.ventiladores().filter((v) => v.encendido).length
  );

  hayAlgoEncendido = computed(
    () => this.consultorio.luces().encendido || this.ventiladoresActivos() > 0
  );

  fotoPerfilUrl = computed(() => {
    const foto = (this.auth.usuarioActual() as any)?.foto_perfil;
    if (!foto || typeof foto !== 'string') return '';
    const limpio = foto.trim();
    if (!limpio || limpio.startsWith('profile-photos/')) return '';
    if (limpio.startsWith('http://') || limpio.startsWith('https://') || limpio.startsWith('data:')) {
      return limpio;
    }
    const base = environment.apiUrl.replace(/\/api\/?$/, '');
    const ruta = limpio.startsWith('/') ? limpio : `/storage/${limpio.replace(/^storage\//, '')}`;
    return `${base}${ruta}`;
  });

  inicialesUsuario = computed(() => {
    const nombre = this.auth.usuarioActual()?.nombre || 'ML';
    const partes = nombre.trim().split(/\s+/);
    return partes
      .slice(0, 2)
      .map((p) => p[0]?.toUpperCase() || '')
      .join('');
  });

  etiquetaRol = computed(() => {
    const u = this.auth.usuarioActual();
    if (u?.rol_etiqueta) return u.rol_etiqueta;
    if (u?.rol === 'super_admin') return 'Doctora titular';
    if (u?.rol === 'doctor') return 'Doctor';
    if (u?.rol === 'asistente') return 'Asistente';
    return 'Personal Autorizado';
  });

  constructor() {
    addIcons({
      logOutOutline,
      peopleOutline,
      powerOutline,
      barChartOutline,
      downloadOutline,
      shieldCheckmarkOutline,
    });
  }

  ngOnInit(): void {
    this.route.data.subscribe((data) => {
      if (data['vista']) {
        this.vistaActiva.set(data['vista']);
      }
    });
  }

  seleccionarPeriodo(rango: RangoEstadistica): void {
    this.consultorio.cambiarRangoEstadisticas(rango);
  }

  onCambiarMes(ev: Event): void {
    const mes = Number((ev.target as HTMLSelectElement).value);
    this.consultorio.cambiarMesAnioEstadisticas(mes, this.consultorio.anioSeleccionado());
  }

  onCambiarAnio(ev: Event): void {
    const anio = Number((ev.target as HTMLSelectElement).value);
    this.consultorio.cambiarMesAnioEstadisticas(this.consultorio.mesSeleccionado(), anio);
  }

  etiquetaPeriodo(): string {
    const r = this.consultorio.rangoEstadistica();
    const anio = this.consultorio.anioSeleccionado();
    if (r === 'semana') return 'Ãšltimos 7 dÃ­as';
    if (r === 'mes') {
      const m = this.mesesLista.find((x) => x.valor === this.consultorio.mesSeleccionado())?.nombre || '';
      return `${m} ${anio}`;
    }
    if (r === 'anio') return `AÃ±o ${anio}`;
    return 'Hoy';
  }

  formatearDuracion(segundos: number): string {
    if (!segundos || segundos <= 0) return '0s';
    const mins = Math.floor(segundos / 60);
    const segs = segundos % 60;
    if (mins === 0) return `${segs}s`;
    return `${mins}m ${segs}s`;
  }

  exportarEstadisticasCsv(): void {
    const r = this.consultorio.rangoEstadistica();
    const sufijo =
      r === 'mes'
        ? `mes-${this.consultorio.mesSeleccionado()}-${this.consultorio.anioSeleccionado()}`
        : r === 'anio'
          ? `anio-${this.consultorio.anioSeleccionado()}`
          : `${r}-${new Date().toISOString().slice(0, 10)}`;

    const filas = [
      ['Asiento', 'Hilera', 'Veces Usado', 'Tiempo Total (seg)', 'Porcentaje Uso (%)'],
      ...this.consultorio.estadisticasAsientos().map((s) => [
        `Silla #${s.numero}`,
        s.hilera,
        s.vecesUsado,
        s.tiempoTotalSegundos,
        `${s.porcentajeUso}%`,
      ]),
    ];
    const csv = filas.map((f) => f.join(',')).join('\n');
    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = `estadisticas-${sufijo}.csv`;
    link.click();
    URL.revokeObjectURL(url);
  }

  salir(): void {
    this.auth.cerrarSesion();
    this.router.navigateByUrl('/login');
  }
}
